// ── Utilities ─────────────────────────────────────────────────────────────
async function api(method, path, body) {
  const opts = { method, headers: { 'Content-Type': 'application/json' } };
  if (body) opts.body = JSON.stringify(body);
  const res = await fetch(path, opts);
  if (!res.ok) throw new Error(`${method} ${path} → ${res.status}`);
  return res.json();
}

function setStatus(msg) {

function escapeHtml(s) {
  return String(s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;')
    .replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

// ── Log stream rendering ──────────────────────────────────────────────────
/**
 * Render an array of log lines into a container.
 * Lines that contain a [REDACTED…] or ****-style mask get visual treatment.
 * @param {HTMLElement} el
 * @param {string[]} lines
 * @param {'before'|'after'} mode  — before highlights; after shows redaction bars
 */
function renderStream(el, lines, mode) {
  if (!lines || lines.length === 0) {
    el.innerHTML = '<span class="log-empty">No log records captured.</span>';
    return;
  }
  el.innerHTML = lines.map((line, i) => {
    let rendered = escapeHtml(line);
    if (mode === 'before') {
      // Highlight canary-format tokens: ****4242, c***@, +92***…567, etc.
      rendered = rendered
        .replace(/(\*{4,}[A-Za-z0-9]{1,6})/g, '<span class="log-highlight">$1</span>')
        .replace(/(c\*{3}@[\w.]+)/g, '<span class="log-highlight">$1</span>')
        .replace(/(\[REDACTED:[^\]]+\])/g, '<span class="log-highlight">$1</span>');
    } else {
      // Show redaction bars for masked values
      rendered = rendered
        .replace(/(\*{4,}[A-Za-z0-9]{1,6})/g, '<span class="log-redacted">$1</span>')
        .replace(/(c\*{3}@[\w.]+)/g, '<span class="log-redacted">$1</span>')
        .replace(/(\[REDACTED:[^\]]+\])/g, '<span class="log-redacted">$1</span>');
    }
    return `<span class="log-line" data-line="${i}">${rendered}</span>`;
  }).join('');
}

// ── Leak table ────────────────────────────────────────────────────────────
function renderLeakTable(leaks) {
  if (!leaks || leaks.length === 0) {
    leakTbody.innerHTML = '<tr><td colspan="7" style="text-align:center;color:var(--muted);padding:20px">No leaks found.</td></tr>';
    return;
  }

  const sorted = [...leaks].sort((a, b) => {
    let va = a[_sortCol], vb = b[_sortCol];
    if (_sortCol === 'severity') { va = SEV_ORDER[va] ?? 9; vb = SEV_ORDER[vb] ?? 9; }
    if (va < vb) return _sortAsc ? -1 : 1;
    if (va > vb) return _sortAsc ?  1 : -1;
    return 0;
  });

  leakTbody.innerHTML = sorted.map(lk => {
    const sevCls  = `sev-${lk.severity}`;
    const fixCls  = `status-${lk.fix_status || 'open'}`;
    const fixLbl  = (lk.fix_status || 'open').charAt(0).toUpperCase() + (lk.fix_status || 'open').slice(1);
    const sel     = lk.fingerprint === _selectedFp ? ' class="selected"' : '';
    return `<tr${sel} data-fp="${escapeHtml(lk.fingerprint)}">
      <td class="${sevCls}">${escapeHtml(lk.severity)}</td>
      <td>${escapeHtml(lk.kind)}</td>
      <td class="file-cell">${escapeHtml(lk.file)}:${lk.line}</td>
      <td>${escapeHtml(lk.sink)}</td>
      <td>${lk.hits}</td>
      <td class="${fixCls}">${fixLbl}</td>
      <td class="file-cell" style="max-width:260px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap">${escapeHtml(lk.sample || '')}</td>
    </tr>`;
  }).join('');

  // Row click
  leakTbody.querySelectorAll('tr[data-fp]').forEach(row => {
    row.addEventListener('click', () => selectLeak(row.dataset.fp));
  });
}

// Column sort headers

    else { _sortCol = col; _sortAsc = true; }

// ── Select leak → show context/diff ──────────────────────────────────────
async function selectLeak(fp) {
  _selectedFp = fp;
  renderLeakTable(_leaks);
  diffPre.innerHTML = '<span class="diff-ctx">Loading…</span>';
  try {
    const ctx = await api('GET', `/api/leaks/${encodeURIComponent(fp)}`);
    renderDiff(ctx);
  } catch (e) {
    diffPre.innerHTML = `<span class="diff-ctx">Error: ${escapeHtml(e.message)}</span>`;
  }
}

function renderDiff(ctx) {
  if (!ctx || !ctx.code) {
    diffPre.innerHTML = '<span class="no-select">No source context available.</span>';
    return;
  }
  const lines = ctx.code.split('\n');
  const html = lines.map(line => {
    const esc = escapeHtml(line);
    if (line.startsWith('>>>')) return `<span class="diff-del">${esc}</span>`;
    return `<span class="diff-ctx">${esc}</span>`;
  }).join('');

  const stratHtml = ctx.strategy
    ? `<span class="diff-hdr">\n# Strategy: ${escapeHtml(ctx.strategy)}\n# Owner: ${escapeHtml(ctx.owner)}</span>\n`
    : '';
  diffPre.innerHTML = stratHtml + html;
}

const GATE_LABELS = {
  no_confirmed_leaks:    'No confirmed leaks remaining',
  no_critical_suspected: 'No critical suspected leaks',

function renderGate(gate) {
  if (!gate) {
    gateList.innerHTML = '<li style="color:var(--muted);font-style:italic">No verification run yet.</li>';
    gateOverall.textContent = '';
    gateOverall.className = 'gate-overall';
    return;
  }
  let delay = 0;
  gateList.innerHTML = Object.entries(GATE_LABELS).map(([key, label]) => {
    const pass = gate.checks && gate.checks[key];
    const icon = pass ? '✓' : '✗';
    const cls  = pass ? 'gate-pass' : 'gate-fail';
    delay += 120;
    return `<li style="animation-delay:${delay}ms">
      <span class="gate-icon ${cls}">${icon}</span>
      <span>
        ${escapeHtml(label)}
        ${!pass && gate.reasons ? `<span class="gate-reason">${escapeHtml(gate.reasons.find(r => r.toLowerCase().includes(key.replace(/_/g,' '))) || '')}</span>` : ''}
      </span>
    </li>`;
  }).join('');

  gateOverall.textContent = gate.passed ? '✓ Gate Passed' : '✗ Gate Failed';
  gateOverall.className   = `gate-overall ${gate.passed ? 'passed' : 'failed'}`;
}

// ── Headline ──────────────────────────────────────────────────────────────
function renderHeadline(before, gate) {
  if (!before) { headlineEl.textContent = 'No scan yet.'; return; }
  const n      = before.leaks ? before.leaks.length : 0;
  const crit   = (before.leaks || []).filter(l => l.severity === 'critical').length;
  const high   = (before.leaks || []).filter(l => l.severity === 'high').length;
  const exec   = (before.executed_sites || []).length;
  const total  = before.total_sites || 0;

  if (gate && gate.passed) {
    headlineEl.textContent = '0 leaks in your logs. Logs still useful.';
  } else {
    headlineEl.textContent = `${n} leak${n !== 1 ? 's' : ''} in your logs`;
  }
  sublineEl.innerHTML =
    `<span class="badge badge-critical">${crit} critical</span>` +
    `<span class="badge badge-high">${high} high</span>` +
    `${exec} of ${total} log lines exercised`;
}

async function followStream(run_id) {
  return new Promise((resolve) => {
    const es = new EventSource(`/api/stream/${run_id}`);
    es.onmessage = e => {
      setStatus(e.data.replace(/^data: /, ''));
      if (e.data.includes('[DONE]') || e.data.includes('DONE') || e.data.includes('[TIMEOUT]')) {
        es.close();
        resolve();
      }
    };
    es.onerror = () => { es.close(); resolve(); };
  });
}

// ── Load all data ─────────────────────────────────────────────────────────
async function loadAll() {
  const [before, leaks, gate] = await Promise.allSettled([
    api('GET', '/api/report?run=before'),
    api('GET', '/api/leaks'),
    api('GET', '/api/gate'),
  ]);

  const beforeData = before.status === 'fulfilled' ? before.value : null;
  const leaksData  = leaks.status  === 'fulfilled' ? leaks.value  : [];
  const gateData   = gate.status   === 'fulfilled' ? gate.value   : null;

  _leaks = leaksData;

  renderHeadline(beforeData, gateData);
  renderLeakTable(_leaks);
  renderGate(gateData);

  // Before log stream from masked log text
  if (beforeData && beforeData.log_text_masked) {
    renderStream(beforeStream, beforeData.log_text_masked.split('\n'), 'before');
  }

  // After log stream
  const afterResult = await api('GET', '/api/report?run=after').catch(() => null);
  if (afterResult && afterResult.log_text_masked) {
    renderStream(afterStream, afterResult.log_text_masked.split('\n'), 'after');
  } else {
    afterStream.innerHTML = '<span class="log-empty">Run Bob fix to see the after state.</span>';
  }

  await loadUnwitnessed();
}

## How LogLeak works

### The core idea

LogLeak exploits the fact that you already have a test suite. Instead of statically scanning source code (which misses runtime formatting like `f"Registered {patient}"` or `%s` args), it **watches the logs your tests actually produce** and checks what's in them. The data flows through five stages:

```
Inject canaries → Capture → Detect → Trace → Redact & Report
```

---

### Stage 1 — Canary injection ([`canaries.py`](logleak/canaries.py))

Seven fake-but-valid PII values are defined:

```python
CANARIES = {
    "email":  "canary.patient@logleak.test",
    "card":   "4242424242424242",          # Luhn-valid
    "cnic":   "35202-1234567-9",
    "phone":  "+923001234567",
    "iban":   "GB82WEST12345698765432",    # mod-97 valid
    "jwt":    "eyJhbGciOiJIUzI1NiJ9...",
    "secret": "llk_canary_9fQ2xV7p...",
}
```

These are injected into the target app's test suite as a `canary` pytest fixture by [`pytest_plugin.py`](logleak/pytest_plugin.py). The tests in `demo/caredesk/tests/` already use `canary["card"]`, `canary["email"]`, etc. when registering patients and charging cards. So the canary values flow through the real app code paths — if a log statement leaks them, we'll see them in the output.

**Why fake-but-valid?** Validators like Luhn (cards) and mod-97 (IBANs) are tested against the canary values, so the detector code is exercised with realistic values. The `.test` TLD ensures the email can never be a real address.

---

### Stage 2 — Capture ([`capture.py`](logleak/capture.py) + [`pytest_plugin.py`](logleak/pytest_plugin.py))

[`scanner.py`](logleak/scanner.py) spawns pytest as a **subprocess** with two things set:
- `LOGLEAK_OUT=/tmp/.../capture.jsonl` — tells the plugin where to write
- `PYTHONPATH=<target>` — makes the app importable

The plugin's `pytest_sessionstart` hook attaches a `LeakCaptureHandler` to the **root logger at DEBUG** before any test runs. Every log record from the app — at any level, from any logger — hits this handler. `handler.emit()` calls `record.getMessage()` which **fully formats the `%s` args** before storing. That's the key: `logger.debug("Charging card %s", card_number)` stores `"Charging card 4242424242424242"`, not the format string.

For `print()`, the plugin wraps `sys.stdout` per-test with `_StdoutTee`. It walks the call stack to find the first frame outside logleak itself, recording the correct `file:line` of the `print()` call.

For exceptions, when `exc_info=True` is set, `formatException()` is called immediately to capture the full traceback text — because that's where `ValueError: Card declined: 4242424242424242` lives.

At session end, all `CapturedRecord` objects are serialised as JSONL to `LOGLEAK_OUT`. Each record carries: `logger`, `level`, `message`, `pathname`, `lineno`, `func_name`, `exc_text`, `sink`.

---

### Stage 3 — Detection ([`detectors.py`](logleak/detectors.py))

`detect(text, canaries)` runs in two passes:

**Pass 1 — exact canary substring search (→ `"confirmed"`)**
For every canary value, it does a plain `text.find(value)` scan. If found, that's a `confirmed` finding. This catches the `secret` canary (`llk_canary_...`) which has no keyword prefix and would never match the pattern-based rules alone.

**Pass 2 — regex pattern matching (→ `"suspected"` unless value is in canary set)**
Seven matchers run in sequence, each with false-positive guards:

| Kind | Guard |
|---|---|
| card | Luhn checksum + 13–19 digits + word boundary |
| IBAN | mod-97 check |
| CNIC plain form | must have the word "cnic" within ±20 chars |
| phone | not preceded by `:` (timestamp guard) |
| secret | entropy ≥ 3.5 and length ≥ 16, or keyword is `password` |
| email/jwt | regex only |

**Overlap resolution:** findings are sorted by start position then by a priority map (`card=7` beats `email=1`). A greedy scan keeps non-overlapping findings, with higher-priority kinds winning ties at the same start position.

---

### Stage 4 — Tracing ([`tracer.py`](logleak/tracer.py))

`trace(records, canaries, root)` groups findings by `(file, line, kind)`. Every record from the same source line that leaks the same kind of PII is counted as one `Leak` with a `hits` counter. The `sample` is the masked version of the message. The fingerprint is `sha1(f"{file}:{line}:{kind}")[:12]` — stable across runs even as line numbers shift.

The `file` path is made relative to the target root for app code, or kept absolute for third-party library paths (anything with `site-packages` in the path). This is how `owner = "third_party"` is determined — Bob gets told to install a filter rather than edit the library.

---

### Stage 5 — Redaction ([`redact.py`](logleak/redact.py))

`mask(kind, value)` applies kind-specific masking rules — never drops anything, always returns a string. Examples:
- card `4242424242424242` → `************4242` (last 4, PCI-compliant)
- email `canary.patient@logleak.test` → `c***@logleak.test` (keeps domain)
- phone `+923001234567` → `+92*******567` (keeps prefix + last 3)
- CNIC → `[REDACTED:cnic]` (last digit encodes gender, so full redaction)

`RedactionFilter` is a `logging.Filter` that rewrites `record.msg`, `record.args`, and `record.exc_text` **before the formatter renders them**. It never returns `False` — it only rewrites, never drops. This is the G6 safety net: even if a fix is missed, the filter catches it at the handler level.

---

### The verification gate ([`verify.py`](logleak/verify.py))

`run_gate(before, after, allowlist_tokens, probe_masked)` compares two `Report` objects. Six boolean checks:

| Check | Logic |
|---|---|
| G1 `no_confirmed_leaks` | `len([l for l in after.leaks if l.confidence == "confirmed"]) == 0` |
| G2 `no_critical_suspected` | No leaks where `severity=="critical"` and `confidence=="suspected"` |
| G3 `tests_pass` | `after.tests_exit_code == 0` |
| G4 `no_over_redaction` | Every token in `["PT-", "APT-", "ORD-"]` still appears in `after.log_text_masked` |
| G5 `logs_retained` | App-owned executed sites after ÷ before ≥ 90% |
| G6 `safety_net` | `probe_safety_net(app_path)` — spawns a subprocess, imports the app's `configure_logging()`, logs the card canary, exits 0 only if the canary doesn't appear in the output |

G4 stops Bob from "fixing" leaks by redacting everything including useful business IDs. G5 stops Bob from deleting log lines entirely. G6 is defence-in-depth — it verifies the `RedactionFilter` is actually wired up in the app's own logging config.

---

### The MCP loop ([`mcp_tools.py`](logleak/mcp_tools.py) + [`mcp_server.py`](logleak/mcp_server.py))

Bob calls four tools via stdio MCP:

```
scan_logs(target)        → runs scan_target(), returns counts by kind/severity
list_leaks(severity?)    → returns Leak dicts, all values passed through sanitize()
get_leak_context(fp)     → returns ±5 lines of source code + fix strategy string
verify_fix(target)       → re-scans workspace, runs gate, returns GateResult
```

`sanitize(obj)` recursively walks the entire return value — dicts, lists, tuples — and calls `redact_text()` on every string. The spec test `test_no_tool_output_contains_raw_canaries` asserts that none of the 7 canary values appear anywhere in any tool output, ever.

The resource `logleak://policy` returns the masking table so Bob uses the same masks as the engine — no guessing about how many digits of a card to show.

---

### Privacy invariant — where raw values stop

```
Raw values exist only here:
  runs/capture.jsonl       (gitignored, subprocess temp dir)

Masked values everywhere else:
  runs/report.json         to_json() calls redact_text()
  runs/leak-report.md      to_markdown() calls mask()
  MCP tool output          sanitize() on every return value
  /api/* responses         _sanitize_json() in server.py
  Bob prompts              leak-report.md is attached, already masked
```

`RedactionFilter` in the fixed app is the last line: even if new code accidentally logs PII after the fix, it gets masked before hitting any handler.

/**
 * Render an array of log lines into a container.
 * Masked tokens produced by logleak/redact.py get visual treatment:
 *
 *   [REDACTED:cnic]          — full redaction block
 *   [REDACTED:jwt]
 *   [REDACTED:secret]
 *   ************4242         — card: leading stars + up to 4 trailing alnum
 *   c***@logleak.test        — email: first char + *** + @domain
 *   +92*******567            — phone: prefix + stars + suffix (all digits/stars)
 *   GB82******************   — IBAN: 4-char prefix + stars
 *
 * @param {HTMLElement} el
 * @param {string[]} lines
 * @param {'before'|'after'} mode  — before=yellow highlight; after=black redaction bar
 */
function renderStream(el, lines, mode) {
  if (!lines || lines.length === 0) {
    el.innerHTML = '<span class="log-empty">No log records captured.</span>';
    return;
  }

  const cls = mode === 'before' ? 'log-highlight' : 'log-redacted';

  // Ordered list of patterns that match every mask format redact.py produces.
  // Each pattern must NOT use capturing groups that interfere with the replace $1.
  const MASK_PATTERNS = [
    // [REDACTED:xxx] — full redaction tokens (cnic, jwt, secret, email fallback, phone fallback)
    /(\[REDACTED:[^\]]+\])/g,
    // card: 8+ stars followed by 1–4 alnum chars (e.g. ************4242)
    /(\*{8,}[A-Za-z0-9]{1,4})/g,
    // email: one char + *** + @ + domain (e.g. c***@logleak.test)
    /([A-Za-z0-9]\*{3}@[A-Za-z0-9.]+\.[A-Za-z]{2,})/g,
    // phone: starts with + or 0, contains digits and stars, ends with digits
    //        e.g. +92*******567  or  0**1234567
    /([+0]\d*\*+\d+)/g,
    // IBAN: 4 uppercase alnum chars followed by 8+ stars (e.g. GB82******************)
    /([A-Z0-9]{4}\*{8,})/g,
    // generic: 4+ consecutive stars anywhere not already caught
    /(\*{4,})/g,
  ];

    let rendered = escapeHtml(line);
    for (const pat of MASK_PATTERNS) {
      rendered = rendered.replace(pat, `<span class="${cls}">$1</span>`);
    }
    return `<span class="log-line" data-line="${i}">${rendered}</span>`;
  }).join('');

## What was wrong and what was fixed

### Bug 1 — Wrong file for the "After" stream (root cause of the screenshot)

`/api/report?run=after` was reading `runs/report.json`, which is **overwritten by every `logleak scan`** — so after running scan it always held the before-state data. The fix runs (`logleak verify`) scanned the workspace into `runs/after/` but never saved the result as a JSON file.

**Fix:** [`mcp_tools.verify_fix()`](logleak/mcp_tools.py) now writes `runs/report_after.json` after every verify run. [`server.py`](logleak/server.py) reads `report_after.json` when it exists, and [`/api/leaks`](logleak/server.py) uses the same file for `fix_status` comparison. The after stream now shows 0 leaks and clean log lines like `Registered patient PT-1001` instead of the full PII dump.

### Bug 2 — Regex patterns too narrow in the frontend

The old patterns used `[A-Za-z0-9]{1,6}` which is too short — `+92*******567` has 3 trailing digits and 7 stars, and `GB82******************` has a long star sequence. The regex `\*{4,}[A-Za-z0-9]{1,6}` also didn't match phone masks at all since they start with `+`.

**Fix:** [`app.js`](ui/app.js) now has six ordered patterns that map exactly to what `redact.py` produces:
- `[REDACTED:xxx]` — full token
- `\*{8,}[A-Za-z0-9]{1,4}` — card (12 stars + last 4)
- `[char]\*{3}@domain` — email
- `[+0]\d*\*+\d+` — phone (`+92*******567`)
- `[A-Z0-9]{4}\*{8,}` — IBAN (`GB82***...`)
- `\*{4,}` — generic fallback

### Bug 3 — `fix_status` was always "open"

`/api/leaks` was comparing before-leaks against `report.json` (also the before data), so no fingerprint was ever missing — everything showed as `open`.

**Fix:** Same as Bug 1 — `/api/leaks` now reads `report_after.json` for the comparison set. All 12 leaks now correctly show `fixed` or `filtered`.
